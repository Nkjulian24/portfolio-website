YoY Growth % =
VAR CurrentYear =
    CALCULATE(
        [Total Revenue],
        DATESYTD('Date'[Date])
    )
VAR PreviousYear =
    CALCULATE(
        [Total Revenue],
        SAMEPERIODLASTYEAR('Date'[Date])
    )
RETURN
    DIVIDE(
        CurrentYear - PreviousYear,
        PreviousYear
    )
